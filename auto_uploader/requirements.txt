# Pre-upload censoring (reuses ../autoreel/) - already installed if you've
# run INSTALL.bat for the main AutoBleepPro project; listed here so this
# folder's requirements.txt is complete on its own too.
#
# faster-whisper is the preferred transcription backend: it runs int8 on
# CPU (several times faster than openai-whisper's fp32) and produces word
# timestamps without torch's DTW pass - which is the code that tries to
# JIT a Triton kernel and warns on every call when there's no CUDA
# toolkit. openai-whisper stays listed as the fallback so existing
# installs keep working.
faster-whisper
openai-whisper

# GPU acceleration for faster-whisper (NVIDIA only). Not installed by
# default because they are large and useless without a card - install
# them yourself, then run `python main.py --gpu-check` to confirm:
#     pip install nvidia-cublas-cu12 nvidia-cudnn-cu12
# On a GPU, large-v3 transcribes a multi-hour stream in a fraction of the
# CPU time, which is what makes the most accurate model practical.
better-profanity
pydub
moviepy

# Kick recording (tools/record_stream.py). Kick is behind Cloudflare, and
# yt-dlp needs a real browser TLS fingerprint to reach it - without this
# every Kick request comes back "HTTP Error 403: Forbidden" and yt-dlp
# warns that "no impersonate target is available". Not needed for YouTube
# or Twitch.
#
# Installed through yt-dlp's own extra rather than as a bare `curl_cffi`,
# because THE VERSION MATTERS and a bare install takes the newest one.
# curl_cffi 0.16 installs and imports perfectly while yt-dlp reports
# every impersonate target as unavailable - which looks like a broken
# install rather than a version mismatch, and cost an evening to find.
# The extra resolves whatever this yt-dlp actually supports (0.15 at time
# of writing) and downgrades if needed:
#
#     python -m pip install -U "yt-dlp[default,curl-cffi]"
#
# Check it worked - every line must NOT say "(unavailable)":
#     python -m yt_dlp --list-impersonate-targets
yt-dlp[default,curl-cffi]
